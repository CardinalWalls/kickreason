AThe MiroMind API call — real, not a wrapper

graph_build.py · narrative.py · run_forecasts.py · prove_kernel.py — all use this stream_call()
One HTTP POST to the hosted deep-research endpoint; we parse the Server-Sent-Events stream of reasoning_steps (thinking · web_search · fetch_url_content) into content + sources + token usage. Stdlib only.
# graph_build.py — the real call (stdlib urllib; SSE stream parse)
API_URL = "https://api.miromind.ai/v1/chat/completions"
MODEL   = "mirothinker-1-7-deepresearch-mini"

def stream_call(messages, timeout):
    body = json.dumps({"model": MODEL, "messages": messages}).encode()
    req = urllib.request.Request(API_URL, data=body, headers={
        "Authorization": f"Bearer {KEY}", "Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        for raw in resp:                          # Server-Sent Events
            obj = json.loads(raw[5:])             # strip "data:"
            delta = obj["choices"][0]["delta"]
            for s in delta.get("reasoning_steps", []):   # <- the reasoning chain
                if s["type"] == "web_search":  ... collect search_results[].url
                elif s["type"] == "fetch_url_content":  ... collect .url + .snippet
            content += delta.get("content", "")   # the judged answer
            usage    = obj.get("usage", usage)    # tokens

DForecast — the de-vig arithmetic (no fake edge)

baseline.py (self-tested) — the parent prob = the sharp market the nodes confirm
# baseline.py — de-vig the closing line (self-tested: prints PASS)
def implied_prob_american(odds):
    return (100/(odds+100)) if odds>0 else ((-odds)/((-odds)+100))
# the trace captured -1000 on USA-advance:
implied_prob_american(-1000) == 1000/1100 == 0.9091  ->  91%
